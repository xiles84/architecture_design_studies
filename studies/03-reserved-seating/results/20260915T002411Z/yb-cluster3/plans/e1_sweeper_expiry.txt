## w_hold_seats
==============================================================================

# The design. Takes only seats whose status says available: an expired hold the
# sweeper has not released yet is NOT free here. The caller rolls back unless
# every seat was taken.

UPDATE event_seat
   SET status = 'held',
       hold_id = $1,
       customer_id = $2,
       hold_expires_at = now() + make_interval(secs => $3::FLOAT8 / 1000.0)
 WHERE event_id = $4
   AND section_no = $5
   AND seat_id = ANY ($6::INT[])
   AND status = 'available'
