# "Is my hold still valid?" before paying: the cart unexpired, every seat still its.

SELECT (SELECT COUNT(*)
          FROM event_seat
         WHERE event_id = $1
           AND section_no = $2
           AND seat_id = ANY ($4::INT[])
           AND hold_id = $3
           AND status = 'held') = cardinality($4::INT[])
