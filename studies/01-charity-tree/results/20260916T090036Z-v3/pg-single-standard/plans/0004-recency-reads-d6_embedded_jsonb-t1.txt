       (e ->> 'a')::BIGINT       AS amount_cents,
       (e ->> 't')::TIMESTAMPTZ  AS donated_at,

 CROSS JOIN LATERAL jsonb_array_elements(p.donations) e
 WHERE p.charity_id = $1
